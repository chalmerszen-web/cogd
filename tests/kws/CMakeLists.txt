cmake_minimum_required(VERSION 3.20)
project(kws_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(ROOT "../.." ABSOLUTE)
set(KWS "${ROOT}/components/kws_c11")
set(KWS_MODEL_SOURCE "${KWS}/generated/probe.c" CACHE FILEPATH "Model for the host shared library")
set(KWS_SECONDARY_SOURCE "" CACHE FILEPATH "Optional independently trained fusion member")
set(SOURCES "${KWS}/kws_quant.c" "${KWS}/kws_detector.c" "${KWS}/kws_nn.c" "${KWS}/kws_fusion.c"
  "${KWS}/kws_frontend.c" "${KWS}/kws_pcen.c" "${KWS}/kws_fft.c" "${KWS}/generated/tables.c")
add_library(kws SHARED ${SOURCES} "${KWS_MODEL_SOURCE}" ${KWS_SECONDARY_SOURCE})
target_include_directories(kws PUBLIC "${KWS}/include" PRIVATE "${ROOT}/third_party/kissfft")
target_compile_options(kws PRIVATE -Wall -Wextra -Werror -O2 -fstack-usage)
target_link_libraries(kws PRIVATE m)
add_executable(kws_test test_kws.c ${SOURCES} "${KWS}/generated/probe.c")
target_include_directories(kws_test PRIVATE "${KWS}/include" "${ROOT}/third_party/kissfft")
target_compile_options(kws_test PRIVATE -Wall -Wextra -Werror -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(kws_test PRIVATE -fsanitize=address,undefined
  -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
target_link_libraries(kws_test PRIVATE m)
add_library(kws_pcen SHARED "${KWS}/kws_pcen.c")
target_include_directories(kws_pcen PUBLIC "${KWS}/include")
target_compile_options(kws_pcen PRIVATE -Wall -Wextra -Werror -O2 -fstack-usage)
add_executable(kws_pcen_test test_pcen.c "${KWS}/kws_pcen.c")
target_include_directories(kws_pcen_test PRIVATE "${KWS}/include")
target_compile_options(kws_pcen_test PRIVATE -Wall -Wextra -Werror -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(kws_pcen_test PRIVATE -fsanitize=address,undefined)
enable_testing()
add_test(NAME kws_sanitized COMMAND kws_test)
add_test(NAME pcen_sanitized COMMAND kws_pcen_test)

set(EL_MODELS "${KWS}/generated/fusion_primary.c" "${KWS}/generated/fusion_secondary_l.c")
add_executable(kws_export_silence "${ROOT}/tools/kws/export_silence.c" ${SOURCES} ${EL_MODELS})
target_include_directories(kws_export_silence PRIVATE "${KWS}/include" "${KWS}" "${ROOT}/third_party/kissfft")
target_compile_definitions(kws_export_silence PRIVATE AGENT_KWS_TRAINED=1 AGENT_KWS_FUSION=1)
target_compile_options(kws_export_silence PRIVATE -Wall -Wextra -Werror -O2)
target_link_libraries(kws_export_silence PRIVATE m)
option(KWS_VERIFY_SILENCE_SEED "Verify checked-in E/L seed hashes" ON)
if(KWS_VERIFY_SILENCE_SEED)
  include("${KWS}/generated/silence_el.cmake")
endif()
add_executable(kws_prime_test test_prime.c ${SOURCES} ${EL_MODELS})
target_include_directories(kws_prime_test PRIVATE "${KWS}/include" "${KWS}" "${ROOT}/third_party/kissfft")
target_compile_definitions(kws_prime_test PRIVATE AGENT_KWS_TRAINED=1 AGENT_KWS_FUSION=1 AGENT_KWS_EL_SILENCE_PRIME=1)
target_compile_options(kws_prime_test PRIVATE -Wall -Wextra -Werror -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(kws_prime_test PRIVATE -fsanitize=address,undefined)
target_link_libraries(kws_prime_test PRIVATE m)
add_test(NAME kws_prime_sanitized COMMAND kws_prime_test)
add_executable(kws_gate_test test_gate.c ${SOURCES} ${EL_MODELS})
target_include_directories(kws_gate_test PRIVATE "${KWS}/include" "${KWS}" "${ROOT}/third_party/kissfft")
target_compile_definitions(kws_gate_test PRIVATE AGENT_KWS_TRAINED=1 AGENT_KWS_FUSION=1 AGENT_KWS_EL_SILENCE_PRIME=1)
target_compile_options(kws_gate_test PRIVATE -Wall -Wextra -Werror -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(kws_gate_test PRIVATE -fsanitize=address,undefined
  -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
target_link_libraries(kws_gate_test PRIVATE m)
add_test(NAME kws_gate_sanitized COMMAND kws_gate_test)
foreach(pair IN ITEMS ef ek)
  set(secondary "${KWS}/generated/fusion_secondary.c")
  if(pair STREQUAL "ek")
    set(secondary "${KWS}/generated/fusion_secondary_k.c")
  endif()
  add_executable(kws_prime_${pair}_test test_prime_fallback.c ${SOURCES} "${KWS}/generated/fusion_primary.c" "${secondary}")
  target_include_directories(kws_prime_${pair}_test PRIVATE "${KWS}/include" "${ROOT}/third_party/kissfft")
  target_compile_definitions(kws_prime_${pair}_test PRIVATE AGENT_KWS_TRAINED=1 AGENT_KWS_FUSION=1)
  target_compile_options(kws_prime_${pair}_test PRIVATE -Wall -Wextra -Werror -O2)
  target_link_libraries(kws_prime_${pair}_test PRIVATE m)
  add_test(NAME kws_prime_${pair}_cold COMMAND kws_prime_${pair}_test)
endforeach()

set(KWS_WIDE_MODEL_SOURCE "" CACHE FILEPATH "Optional untrained48 probe for an isolated host check")
if(KWS_WIDE_MODEL_SOURCE)
  if(NOT EXISTS "${KWS_WIDE_MODEL_SOURCE}")
    message(FATAL_ERROR "The explicit48 probe source is absent")
  endif()
  set(WIDE_SOURCES ${SOURCES})
  list(REMOVE_ITEM WIDE_SOURCES "${KWS}/kws_fusion.c")
  add_library(kws48 SHARED ${WIDE_SOURCES} "${KWS_WIDE_MODEL_SOURCE}")
  target_include_directories(kws48 PUBLIC "${KWS}/include" PRIVATE "${ROOT}/third_party/kissfft")
  target_compile_definitions(kws48 PUBLIC KWS_BUILD_CHANNELS=48)
  target_compile_options(kws48 PRIVATE -Wall -Wextra -Werror -Wpedantic -O2 -fstack-usage)
  target_link_libraries(kws48 PRIVATE m)
  add_executable(kws_wide_test test_wide.c ${WIDE_SOURCES} "${KWS_WIDE_MODEL_SOURCE}")
  target_include_directories(kws_wide_test PRIVATE "${KWS}/include" "${KWS}" "${ROOT}/third_party/kissfft")
  target_compile_definitions(kws_wide_test PRIVATE KWS_BUILD_CHANNELS=48)
  target_compile_options(kws_wide_test PRIVATE -Wall -Wextra -Werror -Wpedantic -O1 -g
    -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(kws_wide_test PRIVATE -fsanitize=address,undefined
    -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
  target_link_libraries(kws_wide_test PRIVATE m)
  add_test(NAME kws_wide_sanitized COMMAND kws_wide_test)
endif()
