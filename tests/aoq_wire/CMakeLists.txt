cmake_minimum_required(VERSION 3.22)
project(cogd_aoq_wire_tests LANGUAGES CXX)
enable_testing()
get_filename_component(AOQ "${CMAKE_CURRENT_LIST_DIR}/../../ESP32-P4-WIFI6-Touch-LCD-3.5/components/p4_aoq" ABSOLUTE)
foreach(test setup objects window media_control goaway track_profiles datagram video)
  add_executable(test_${test} test_${test}.cpp "${AOQ}/aoq_wire.cpp" "${AOQ}/aoq_objects.cpp" "${AOQ}/aoq_media_control.cpp" "${AOQ}/aoq_video.cpp")
  target_include_directories(test_${test} PRIVATE "${AOQ}")
  target_compile_features(test_${test} PRIVATE cxx_std_17)
  if(MSVC)
    target_compile_options(test_${test} PRIVATE /W4 /WX /UNDEBUG)
  else()
    target_compile_options(test_${test} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME aoq_${test} COMMAND test_${test})
endforeach()
find_package(Threads REQUIRED)
add_executable(test_packet_queue test_packet_queue.cpp)
target_include_directories(test_packet_queue PRIVATE "${AOQ}")
target_compile_features(test_packet_queue PRIVATE cxx_std_17)
target_link_libraries(test_packet_queue PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(test_packet_queue PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(test_packet_queue PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME aoq_packet_queue COMMAND test_packet_queue)
