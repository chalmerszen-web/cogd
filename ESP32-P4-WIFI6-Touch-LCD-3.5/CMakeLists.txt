cmake_minimum_required(VERSION 3.22)
get_filename_component(COGD_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(IDF_TARGET esp32p4 CACHE STRING "Target")
set(AGENT_BOARD_P4 ON CACHE BOOL "Waveshare P4 board")
option(AGENT_P4_WAKENET10 "Official WakeNet10 nihaoxiaozhi backend" ON)
option(AGENT_P4_AOQ_PROBE "Experimental native AOQ connectivity probe" OFF)
set(COGD_COMM_MODE "WSS" CACHE STRING "COGD realtime communication backend: WSS or AOQ")
set_property(CACHE COGD_COMM_MODE PROPERTY STRINGS WSS AOQ)
if(NOT COGD_COMM_MODE STREQUAL "WSS" AND NOT COGD_COMM_MODE STREQUAL "AOQ")
  message(FATAL_ERROR "COGD_COMM_MODE must be WSS or AOQ")
endif()
if(COGD_COMM_MODE STREQUAL "AOQ")
  if(NOT SDKCONFIG_DEFAULTS)
    set(SDKCONFIG_DEFAULTS "sdkconfig.defaults")
  endif()
  list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.aoq.defaults")
set(PROJECT_VER "0.10.9-p4-noise-tail")
elseif(AGENT_P4_AOQ_PROBE)
  set(PROJECT_VER "0.9.8-p4-aoq-probe")
elseif(AGENT_P4_WAKENET10)
  set(PROJECT_VER "0.9.7-p4-tutor")
elseif(AGENT_P4_KWS_MEGA AND AGENT_P4_KWS_NS)
  set(PROJECT_VER "0.4.1-p4-wake96ns")
elseif(AGENT_P4_KWS_MEGA)
  set(PROJECT_VER "0.4.0-p4-wake96")
elseif(AGENT_P4_KWS_SYNC_CONFIRM)
  set(PROJECT_VER "0.3.1-p4-wake48sync")
elseif(AGENT_P4_KWS_WIDE)
  set(PROJECT_VER "0.3.0-p4-wake48")
else()
  set(PROJECT_VER "0.7.0-p4-interaction")
endif()
set(EXTRA_COMPONENT_DIRS "${COGD_ROOT}/main" "${COGD_ROOT}/components")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
idf_build_set_property(AGENT_P4_AOQ_PROBE "${AGENT_P4_AOQ_PROBE}")
idf_build_set_property(COGD_COMM_MODE "${COGD_COMM_MODE}")
project(cogd_p4)
# Adapter 0.6 uses LVGL 9.5's supported forwarding headers, which emit a
# deprecation #warning. Keep all other errors enabled and scope this to it.
target_compile_options(__idf_espressif__esp_lvgl_adapter PRIVATE -Wno-error=cpp)
if(AGENT_TEXT_PREFETCH)
  include("${COGD_ROOT}/cmake/http_preconnect.cmake")
  agent_configure_http_preconnect()
endif()
